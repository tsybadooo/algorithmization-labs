1 Програма lab_work_02_01 — параметри циліндра

Програму з методичних вказівок введено, відлагоджено та протестовано. Під час налагодження виявлено і виправлено помилки в наведеному тексті: площа повної поверхні обчислювалась як sb + 2πr²h (зайвий множник h), а об’єм — як πrh²; правильні формули sp = sb + 2πr², v = πr²h. Застарілі заголовки <iostream.h>, <conio.h> і функції clrscr()/getch() замінено засобами стандартного С++. Додано перевірку коректності введення.

Файл lab_work_02_01.cpp
/*****************************************************/
/*  lab_work_02_01.                                  */
/*  Лінійні обчислювальні процеси засобами мови С++  */
/*  Дано: r - радіус основи прямого кругового        */
/*  циліндра; h - висота циліндра.                   */
/*  Обчислити: sb = 2*pi*r*h - площу бічної поверхні;*/
/*  sp = sb + 2*pi*r*r - площу повної поверхні;      */
/*  v = pi*r*r*h - об'єм циліндра. Результати подати */
/*  у вигляді таблиці.                               */
/*  Під час налагодження виправлено помилки методички:*/
/*  sp=sb+2*pi*r*r*h -> sp=sb+2*pi*r*r;              */
/*  v=pi*r*h*h       -> v=pi*r*r*h;                  */
/*  <iostream.h>, clrscr(), getch() -> стандарт C++. */
/*****************************************************/
#include <iostream>
#include <iomanip>
#include <cmath>
#include "../common_console.h"
using namespace std;

int main()
{
    console_utf8();
    const double pi = 3.14159265358979; // константа pi
    double r;   // радіус основи циліндра
    double h;   // висота циліндра
    double sb;  // площа бічної поверхні
    double sp;  // площа повної поверхні
    double v;   // об'єм циліндра

    // Введення заданих величин
    cout << " Введіть радіус основи циліндра: "; cin >> r;
    cout << " Введіть висоту циліндра: ";        cin >> h;
    if (!cin || r <= 0 || h <= 0) { cout << " Помилка: r та h мають бути додатними числами\n"; return 1; }

    // Обчислення параметрів циліндра
    sb = 2 * pi * r * h;
    sp = sb + 2 * pi * r * r;
    v  = pi * r * r * h;

    // Друк результатів у вигляді таблиці
    cout << fixed << setprecision(3);
    cout << "\n Вхідні величини та результати обчислень:\n";
    cout << " +----------------------------+------------+\n";
    cout << " |        Вхідні величини     |            |\n";
    cout << " +----------------------------+------------+\n";
    cout << " | Радіус основи циліндра     |" << setw(11) << r  << " |\n";
    cout << " | Висота циліндра            |" << setw(11) << h  << " |\n";
    cout << " +----------------------------+------------+\n";
    cout << " |    Результати обчислень    |            |\n";
    cout << " +----------------------------+------------+\n";
    cout << " | Бічна поверхня циліндра    |" << setw(11) << sb << " |\n";
    cout << " | Повна поверхня циліндра    |" << setw(11) << sp << " |\n";
    cout << " | Об'єм циліндра             |" << setw(11) << v  << " |\n";
    cout << " +----------------------------+------------+\n";
    return 0;
}

2 Програма lab_work_02_02 — бібліотека математичних функцій

У тексті методички вирази pow(a3, 1/2) і pow(b3, 1/3) завжди дорівнюють 1, оскільки 1/2 і 1/3 — цілочисельне ділення, що дає 0. Їх замінено на sqrt(a3) і cbrt(b3). Через це результати відрізняються від наведених у методичці (там a4 = b4 = 1,00).

Файл lab_work_02_02.cpp
/*****************************************************/
/* lab_work_02_02.                                   */
/*  Бібліотека арифметичних функцій С++.             */
/*  Дано: x,y,z - змінні дійсного типу.              */
/*  Обчислити a=f1(x,y,z) та b=f2(x,y,z), вивести    */
/*  вхідні величини, проміжні величини і результати  */
/*  у вигляді таблиці.                               */
/*  Виправлено помилки методички: pow(a3,1/2) та     */
/*  pow(b3,1/3) давали pow(..,0)=1 через цілочисельне*/
/*  ділення; замінено на sqrt() та cbrt().           */
/*****************************************************/
#include <iostream>
#include <iomanip>
#include <cmath>
#include "../common_console.h"
using namespace std;

int main()
{
    console_utf8();
    const double pi = 3.14159265358979;
    double x, y, z;
    double a1, a2, a3, a4, a5, a;
    double b1, b2, b3, b4, b5, b;

    cout << " Введіть значення трьох дійсних змінних x y z:\n "; cin >> x >> y >> z;
    if (!cin) { cout << " Помилка введення\n"; return 1; }

    // Обчислення виразу a
    a1 = y / (2 + y * y + fabs(y * z));
    a2 = acos(a1);
    a3 = pi + x * x + pow(sin(y * z), 2);
    a4 = sqrt(a3);                       // було pow(a3, 1/2)
    a5 = log10(2 / (x * x + y * y + 1));
    a  = a2 * (a4 - a5);

    // Обчислення виразу b
    b1 = exp(-3 * x * x + y / 3 - z);
    b2 = pow(tan(x), 2);
    b3 = b2 / (1 + sin(pi * y / 2));
    b4 = cbrt(b3);                       // було pow(b3, 1/3)
    b5 = (x + y + z) / (x * y * z);
    b  = b1 + b4 - pow(b5, 2);

    cout << fixed << setprecision(3);
    cout << "\n Вхідні величини та результати обчислень:\n";
    cout << " +-----------------------------------+\n";
    cout << " |          Вхідні величини          |\n";
    cout << " +-----------------------------------+\n";
    cout << " | x = " << setw(10) << x << "                    |\n";
    cout << " | y = " << setw(10) << y << "                    |\n";
    cout << " | z = " << setw(10) << z << "                    |\n";
    cout << " +-----------------------------------+\n";
    cout << " |        Результати обчислень       |\n";
    cout << " +-----------------+-----------------+\n";
    cout << " | a1 = " << setw(10) << a1 << " | b1 = " << setw(10) << b1 << " |\n";
    cout << " | a2 = " << setw(10) << a2 << " | b2 = " << setw(10) << b2 << " |\n";
    cout << " | a3 = " << setw(10) << a3 << " | b3 = " << setw(10) << b3 << " |\n";
    cout << " | a4 = " << setw(10) << a4 << " | b4 = " << setw(10) << b4 << " |\n";
    cout << " | a5 = " << setw(10) << a5 << " | b5 = " << setw(10) << b5 << " |\n";
    cout << " | a  = " << setw(10) << a  << " | b  = " << setw(10) << b  << " |\n";
    cout << " +-----------------+-----------------+\n";
    return 0;
}

4 Індивідуальне завдання (варіант 19)

Обчислити значення a і b при заданих x, y, z:

a = cos² [ (π − ln(1 + 3/(x² + 2)) + |x + y/2| − (x − 4,2)(y + 2,8)) / ((2x² + 3y² + ln(1 + |z|)) · sin((x + y)² + 1)) ];

b = √( lg(25 + (3y² + x² + 1,43)/(4 + sin(x/2))) · (4 + arctg((y − 0,1x)/(z² + 8))) ).

Аналіз області визначення: знаменник у b завжди додатний (4 + sin(x/2) ≥ 3, z² + 8 ≥ 8), аргумент lg більший за 25, а 4 + arctg(...) > 0, тому b визначене при будь-яких x, y, z. Вираз a не визначений, якщо знаменник дорівнює нулю, — цей випадок програма перевіряє.

Рисунок 1 – Блок-схема алгоритму індивідуального завдання

Файл lab_work_02_ind.cpp
/*****************************************************************/
/* lab_work_02_ind. Індивідуальне завдання, варіант 19.          */
/* Обчислити при заданих x, y, z:                                */
/*        pi - ln(1 + 3/(x^2+2)) + |x + y/2| - (x-4.2)(y+2.8)    */
/* a = cos^2 ---------------------------------------------------- */
/*         (2x^2 + 3y^2 + ln(1+|z|)) * sin((x+y)^2 + 1)          */
/*                                                               */
/* b = sqrt( lg(25 + (3y^2+x^2+1.43)/(4+sin(x/2)))               */
/*           * (4 + arctg((y-0.1x)/(z^2+8))) )                   */
/*****************************************************************/
#include <iostream>
#include <iomanip>
#include <cmath>
#include "../common_console.h"
using namespace std;

int main()
{
    console_utf8();
    const double pi = 3.14159265358979;
    double x, y, z;
    cout << " Введіть x, y, z: "; cin >> x >> y >> z;
    if (!cin) { cout << " Помилка введення\n"; return 1; }

    // ---- обчислення a через проміжні величини ----
    double a1 = pi - log(1 + 3 / (x * x + 2));            // pi - ln(1+3/(x^2+2))
    double a2 = fabs(x + y / 2);                          // |x + y/2|
    double a3 = (x - 4.2) * (y + 2.8);                    // (x-4.2)(y+2.8)
    double a4 = 2 * x * x + 3 * y * y + log(1 + fabs(z)); // перший множник знаменника
    double a5 = sin(pow(x + y, 2) + 1);                   // sin((x+y)^2+1)
    double den = a4 * a5;
    if (fabs(den) < 1e-12) { cout << " Знаменник дорівнює нулю - a не визначено\n"; return 1; }
    double a = pow(cos((a1 + a2 - a3) / den), 2);

    // ---- обчислення b ----
    double b1 = 25 + (3 * y * y + x * x + 1.43) / (4 + sin(x / 2)); // 4+sin>=3, ділення безпечне
    double b2 = log10(b1);
    double b3 = 4 + atan((y - 0.1 * x) / (z * z + 8));
    double b  = sqrt(b2 * b3);                            // b2>1, b3>0 - корінь визначений

    cout << fixed << setprecision(5);
    cout << "\n +------------------------------------------+\n";
    cout << " |  x = " << setw(12) << x << "                        |\n";
    cout << " |  y = " << setw(12) << y << "                        |\n";
    cout << " |  z = " << setw(12) << z << "                        |\n";
    cout << " +--------------------+---------------------+\n";
    cout << " | a1 = " << setw(13) << a1 << " | b1 = " << setw(14) << b1 << " |\n";
    cout << " | a2 = " << setw(13) << a2 << " | b2 = " << setw(14) << b2 << " |\n";
    cout << " | a3 = " << setw(13) << a3 << " | b3 = " << setw(14) << b3 << " |\n";
    cout << " | a4 = " << setw(13) << a4 << " |                     |\n";
    cout << " | a5 = " << setw(13) << a5 << " |                     |\n";
    cout << " +--------------------+---------------------+\n";
    cout << " | a  = " << setw(13) << a  << " | b  = " << setw(14) << b  << " |\n";
    cout << " +--------------------+---------------------+\n";
    return 0;
}
