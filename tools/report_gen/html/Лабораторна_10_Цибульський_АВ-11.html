/* Вправа 1. Статистика рядка (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main(void) {
    console_utf8();
    char s[256];
    printf("Введіть рядок: ");
    fgets(s, sizeof(s), stdin);
    s[strcspn(s,"\n")] = '\0';
    int letters=0,digits=0,spaces=0,upper=0,lower=0,other=0;
    int freq[256]={0};
    for (int i=0; s[i]; i++) {
        unsigned char c=s[i];
        freq[c]++;
        if      (isupper(c)){upper++;letters++;}
        else if (islower(c)){lower++;letters++;}
        else if (isdigit(c)) digits++;
        else if (isspace(c)) spaces++;
        else                 other++;
    }
    printf("Довжина: %zu\n",strlen(s));
    printf("Літери: %d (вел: %d, мал: %d)\n",letters,upper,lower);
    printf("Цифри: %d, Пробіли: %d, Інші: %d\n",digits,spaces,other);
    /* Паліндром? */
    int n=strlen(s),ok=1;
    for(int i=0;i<n/2;i++)
        if(tolower(s[i])!=tolower(s[n-1-i])){ok=0;break;}
    printf("Паліндром: %s\n",ok?"так":"ні");
    /* Символи з частотою >= 2 */
    printf("Частотні символи (>=2):\n");
    for(int c=32;c<127;c++)
        if(freq[c]>=2) printf("  '%c': %d\n",c,freq[c]);
    return 0;
}

/* Вправа 2. Шифр Цезаря та обробка слів (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <string.h>
#include <ctype.h>
void caesar(char *dst, const char *src, int shift) {
    shift=((shift%26)+26)%26;
    for(int i=0;src[i];i++){
        char c=src[i];
        if(isupper(c))      dst[i]='A'+(c-'A'+shift)%26;
        else if(islower(c)) dst[i]='a'+(c-'a'+shift)%26;
        else                dst[i]=c;
    }
    dst[strlen(src)]='\0';
}
int count_words(const char *s){
    int count=0,in_word=0;
    for(int i=0;s[i];i++){
        if(!isspace(s[i])){if(!in_word){count++;in_word=1;}}
        else in_word=0;
    }
    return count;
}
void reverse_words(char *s){
    /* Перевернути весь рядок */
    int n=strlen(s),l=0,r=n-1;
    while(l<r){char t=s[l];s[l++]=s[r];s[r--]=t;}
    /* Перевернути кожне слово */
    l=0;
    for(int i=0;i<=n;i++){
        if(i==n||isspace(s[i])){
            int ll=l,rr=i-1;
            while(ll<rr){char t=s[ll];s[ll++]=s[rr];s[rr--]=t;}
            l=i+1;
        }
    }
}
int main(void) {
    console_utf8();
    char plain[256],enc[256],dec[256];
    int shift;
    printf("Рядок: "); fgets(plain,sizeof(plain),stdin);
    plain[strcspn(plain,"\n")]='\0';
    printf("Зсув: "); scanf("%d",&shift);
    caesar(enc,plain,shift);
    caesar(dec,enc,-shift);
    printf("Оригінал:    %s\n",plain);
    printf("Зашифровано: %s\n",enc);
    printf("Розшифровано:%s\n",dec);
    printf("Слів у рядку: %d\n",count_words(plain));
    char buf[256]; strcpy(buf,plain);
    reverse_words(buf);
    printf("Слова навпаки: %s\n",buf);
    return 0;
}

/* Власна strlen */
size_t my_strlen(const char *s){
    const char *p=s; while(*p) p++; return p-s;
}
/* Власна strcmp */
int my_strcmp(const char *a, const char *b){
    while(*a&&*a==*b){a++;b++;}
    return (unsigned char)*a-(unsigned char)*b;
}
/* Перевірка балансу дужок */
int check_brackets(const char *s){
    char stack[256]; int top=0;
    for(int i=0;s[i];i++){
        char c=s[i];
        if(c=='('||c=='['||c=='{') stack[top++]=c;
        else if(c==')'||c==']'||c=='}'){
            if(top==0) return 0;
            char open=stack[--top];
            if((c==')'&&open!='(')||(c==']'&&open!='[')||(c=='}'&&open!='{'))
                return 0;
        }
    }
    return top==0;
}
int main(void) {
    console_utf8();
    /* Тест власних функцій */
    const char *s="Hello World";
    printf("strlen=%zu  my_strlen=%zu\n",strlen(s),my_strlen(s));
    printf("strcmp: %d  my_strcmp: %d\n",strcmp("abc","abd"),my_strcmp("abc","abd"));
    /* Тест балансу дужок */
    const char *tests[]={
        "({[]})","([)]","{[()()]}","(((",""};
    for(int i=0;tests[i][0]||i==4;i++){
        if(i==4) break;
        printf("\"%s\": %s\n",tests[i],
               check_brackets(tests[i])?"збалансовано":"незбалансовано");
    }
    return 0;
}

2.1 Завдання 19 (таблиця 3.1)

Ввести рядок. Замінити всі числа у рядку на їх квадрати. Наприклад: «abc 3 def 5» стає «abc 9 def 25».

Рядок читається функцією fgets(), символ нового рядка вилучається через strcspn(). Під час проходу по рядку початком числа вважається цифра, перед якою немає літери чи цифри (isdigit, isalnum), або знак «−» на початку рядка чи після пропуску. Значення числа накопичується в long long, квадрат дописується в результат функцією sprintf(). Послідовності на зразок «x2» або «12abc» є частиною слова і не змінюються.

Файл lab10_var19_t1.c
/*
 * Лабораторна робота №10. Індивідуальне завдання №19 (таблиця 3.1).
 * Ввести рядок. Замінити всі числа у рядку на їх квадрати.
 * Приклад: 'abc 3 def 5' -> 'abc 9 def 25'.
 *
 * Алгоритм:
 *  - рядок читається fgets(), символ '\n' прибирається;
 *  - прохід по рядку: якщо поточний символ - цифра (isdigit) і перед
 *    ним не літера/цифра, то це початок числа; враховується знак '-',
 *    якщо перед ним пропуск або початок рядка;
 *  - цифри числа накопичуються у long long, квадрат дописується у
 *    результуючий рядок через sprintf; інші символи копіюються як є;
 *  - числа, "приклеєні" до слів (напр. 'x2'), числами не вважаються.
 */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include "../common_console.h"

#define LEN 256

int main(void)
{
    console_utf8();
    char s[LEN], res[LEN * 4] = "";
    size_t rlen = 0;
    int count = 0;

    printf("Введіть рядок: ");
    if (fgets(s, sizeof s, stdin) == NULL) { printf("Помилка читання\n"); return 1; }
    s[strcspn(s, "\n")] = '\0';                     /* прибрати символ нового рядка */

    for (size_t i = 0; s[i] != '\0'; ) {
        unsigned char c = (unsigned char)s[i];
        unsigned char prev = i > 0 ? (unsigned char)s[i - 1] : ' ';
        int neg = (c == '-' && isdigit((unsigned char)s[i + 1]) && (i == 0 || isspace(prev)));
        if ((isdigit(c) && !isalnum(prev)) || neg) {
            size_t j = i + (neg ? 1 : 0);
            long long v = 0;
            while (isdigit((unsigned char)s[j]) && v < 100000000LL) v = v * 10 + (s[j++] - '0');
            if (isalpha((unsigned char)s[j]) || isdigit((unsigned char)s[j])) {
                /* число переходить у слово або занадто довге - копіюємо без змін */
                while (s[i] != '\0' && !isspace((unsigned char)s[i])) res[rlen++] = s[i++];
                continue;
            }
            if (neg) v = -v;
            rlen += (size_t)sprintf(res + rlen, "%lld", v * v);   /* дописати квадрат */
            count++;
            i = j;
        } else {
            res[rlen++] = s[i++];                    /* звичайний символ */
        }
        res[rlen] = '\0';
    }
    printf("Результат:     %s\n", res);
    printf("Замінено чисел: %d\n", count);
    return 0;
}

2.2 Завдання 19 (таблиця 3.2)

Ввести рядок з числом у римській системі. Перетворити у арабське число. Підтримати I, V, X, L, C, D, M.

Пропуски вилучаються, літери переводяться у верхній регістр (toupper). Прохід зліва направо: якщо значення поточної цифри менше за наступну, воно віднімається (IV = 5 − 1), інакше додається. Щоб відкинути неправильні записи (IIII, IC, VX), отримане число переводиться назад у канонічний римський запис жадібним алгоритмом і порівнюється з введеним (strcmp).

Рисунок 1 – Блок-схема перетворення римського числа

Файл lab10_var19_t2.c
/*
 * Лабораторна робота №10. Індивідуальне завдання №19 (таблиця 3.2).
 * Ввести рядок з числом у римській системі. Перетворити у арабське
 * число. Підтримати I, V, X, L, C, D, M.
 *
 * Алгоритм:
 *  - рядок читається fgets(), пропуски на краях відкидаються,
 *    літери переводяться у верхній регістр (toupper);
 *  - прохід зліва направо: якщо значення поточної цифри менше за
 *    наступну - воно віднімається (IV = 5 - 1), інакше додається;
 *  - для перевірки коректності отримане число переводиться назад
 *    у римський запис (функція to_roman) і порівнюється з введеним:
 *    так відкидаються записи на зразок IIII, VX, IC, MMMM.
 */
#include <stdio.h>
#include <string.h>
#include <ctype.h>
#include "../common_console.h"

int roman_value(char c);
int roman_to_int(const char *s);
void to_roman(int n, char *out);

int main(void)
{
    console_utf8();
    char line[256], s[256], back[32];
    printf("Введіть римське число: ");
    if (fgets(line, sizeof line, stdin) == NULL) { printf("Помилка читання\n"); return 1; }

    /* вилучити пропуски та перевести у верхній регістр */
    size_t k = 0;
    for (size_t i = 0; line[i] != '\0'; i++)
        if (!isspace((unsigned char)line[i])) s[k++] = (char)toupper((unsigned char)line[i]);
    s[k] = '\0';

    if (k == 0) { printf("Помилка: порожній рядок\n"); return 1; }
    int v = roman_to_int(s);
    if (v <= 0) { printf("Помилка: '%s' містить недопустимі символи\n", s); return 1; }
    to_roman(v, back);
    if (strcmp(back, s) != 0) {
        printf("Помилка: '%s' не є правильним римським записом (правильно для %d: %s)\n", s, v, back);
        return 1;
    }
    printf("%s = %d\n", s, v);
    return 0;
}

/* Значення однієї римської цифри або 0, якщо символ недопустимий. */
int roman_value(char c)
{
    switch (c) {
        case 'I': return 1;    case 'V': return 5;
        case 'X': return 10;   case 'L': return 50;
        case 'C': return 100;  case 'D': return 500;
        case 'M': return 1000; default:  return 0;
    }
}

/* Перетворення римського запису в ціле; повертає -1 для недопустимих символів. */
int roman_to_int(const char *s)
{
    int total = 0;
    size_t n = strlen(s);
    for (size_t i = 0; i < n; i++) {
        int cur = roman_value(s[i]);
        if (cur == 0) return -1;
        int next = (i + 1 < n) ? roman_value(s[i + 1]) : 0;
        if (cur < next) total -= cur; else total += cur;
    }
    return total;
}

/* Канонічний римський запис числа 1..3999 (жадібний алгоритм). */
void to_roman(int n, char *out)
{
    static const int val[13] = {1000, 900, 500, 400, 100, 90, 50, 40, 10, 9, 5, 4, 1};
    static const char *sym[13] = {"M", "CM", "D", "CD", "C", "XC", "L", "XL", "X", "IX", "V", "IV", "I"};
    out[0] = '\0';
    if (n <= 0 || n > 3999) return;
    for (int i = 0; i < 13; i++)
        while (n >= val[i]) { strcat(out, sym[i]); n -= val[i]; }
}

Висновки

Засвоєно подання рядків у С (масив char з нульовим термінатором), безпечне введення fgets(), функції <string.h> (strlen, strcmp, strcpy, strcat, strcspn, strtok) і <ctype.h>. Індивідуальні програми протестовано на коректних, граничних і помилкових даних.

Відповіді на контрольні питання

1. Як рядок представлений у пам’яті мови С? Що таке нульовий термінатор? Масив char, що закінчується символом '\0' (нульовий термінатор), який позначає кінець рядка.

2. Чому не можна порівнювати рядки через ==? Яку функцію слід використовувати? == порівнює адреси масивів, а не вміст; слід використовувати strcmp().

3. Яка різниця між sizeof(s) та strlen(s) для масиву char s[20]? sizeof(s) = 20 — розмір масиву; strlen(s) — кількість символів до '\0'.

4. Поясніть роботу strtok(). Чому перший виклик і наступні відрізняються? strtok замінює роздільники на '\0' і повертає черговий токен; при першому виклику передається рядок, далі — NULL, бо функція запам’ятовує позицію у внутрішньому статичному стані.

5. Як безпечно скопіювати рядок, уникаючи переповнення буфера? snprintf(dst, sizeof dst, "%s", src) або strncpy з явним встановленням dst[size−1] = '\0'; попередньо перевіряти довжину.

6. Що повертає strcmp(s1, s2) для рівних рядків та для s1 < s2? 0 для рівних, від’ємне значення, якщо s1 лексикографічно менший.

7. Як видалити символ \n із рядка, прочитаного через fgets()? s[strcspn(s, "\n")] = '\0';

8. Що таке isalpha(), isdigit(), isspace()? Для якого типу аргументу їх слід викликати? Перевірки, чи є символ літерою, цифрою, пробільним символом. Аргумент має бути значенням unsigned char (або EOF), тому char приводять: isdigit((unsigned char)c).

9. Як знайти кількість входжень символу у рядку без <string.h>? Цикл до '\0': for (n = 0; *s; s++) if (*s == c) n++;

10. Поясніть принцип шифру Цезаря. Як коректно обробити від’ємний зсув? Кожна літера зсувається на k позицій алфавіту по колу: (c − 'a' + k) mod 26. Для від’ємного k використовують ((x % 26) + 26) % 26.
