#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define N 10
void print_arr(int a[], int n) {
    printf("[ ");
    for (int i = 0; i < n; i++) printf("%4d", a[i]);
    printf(" ]\n");
}
void bubble_sort(int a[], int n) {
    for (int i = 0; i < n-1; i++)
        for (int j = 0; j < n-i-1; j++)
            if (a[j] > a[j+1]) {
                int t = a[j]; a[j] = a[j+1]; a[j+1] = t;
            }
}
int main(void) {
    console_utf8();
    srand((unsigned)time(NULL));
    int a[N];
    /* Заповнення випадковими числами від -50 до 50 */
    for (int i = 0; i < N; i++)
        a[i] = rand() % 101 - 50;
    printf("Масив:  "); print_arr(a, N);
    /* Статистика */
    int sum = 0, mn = a[0], mx = a[0];
    int neg = 0, pos = 0, zero = 0;
    for (int i = 0; i < N; i++) {
        sum += a[i];
        if (a[i] < mn) mn = a[i];
        if (a[i] > mx) mx = a[i];
        if      (a[i] < 0) neg++;
        else if (a[i] > 0) pos++;
        else               zero++;
    }
    printf("Сума: %d,  Середнє: %.2f\n", sum, (double)sum/N);
    printf("Мін: %d,  Макс: %d\n", mn, mx);
    printf("Від'ємних: %d, Нулів: %d, Додатніх: %d\n",
           neg, zero, pos);
    /* Сортування */
    bubble_sort(a, N);
    printf("Sorted: "); print_arr(a, N);
    /* Медіана */
    double median = (N % 2 == 0)
        ? (a[N/2-1] + a[N/2]) / 2.0
        : a[N/2];
    printf("Медіана: %.1f\n", median);
    return 0;
}

/* Вправа 2. Пошук у масиві: лінійний та бінарний (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define N 15
int linear_search(int a[], int n, int x) {
    for (int i = 0; i < n; i++)
        if (a[i] == x) return i;
    return -1;
}
int binary_search(int a[], int n, int x) {
    int lo = 0, hi = n - 1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        printf("  Перевіряємо a[%d]=%d\n", mid, a[mid]);
        if      (a[mid] == x) return mid;
        else if (a[mid] < x)  lo = mid + 1;
        else                  hi = mid - 1;
    }
    return -1;
}
void qsort_arr(int a[], int n) {  /* простий qsort через stdlib */
    /* використаємо бульбашку для простоти */
    for (int i=0;i<n-1;i++)
        for (int j=0;j<n-i-1;j++)
            if(a[j]>a[j+1]){int t=a[j];a[j]=a[j+1];a[j+1]=t;}
}

/* Вправа 3. Операції з матрицею (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define ROWS 4
#define COLS 4
void print_matrix(int a[][COLS], int r, int c) {
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++)
            printf("%5d", a[i][j]);
        printf("\n");
    }
}
int main(void) {
    console_utf8();
    srand(7);
    int m[ROWS][COLS];
    /* Заповнення */
    for (int i = 0; i < ROWS; i++)
        for (int j = 0; j < COLS; j++)
            m[i][j] = rand() % 20 - 5;
    printf("Матриця %dx%d:\n", ROWS, COLS);
    print_matrix(m, ROWS, COLS);
    /* Суми рядків */
    printf("\nСуми рядків: ");
    for (int i = 0; i < ROWS; i++) {
        int s = 0;
        for (int j = 0; j < COLS; j++) s += m[i][j];
        printf("%d ", s);
    }
    /* Суми стовпців */
    printf("\nСуми стовп.: ");
    for (int j = 0; j < COLS; j++) {
        int s = 0;
        for (int i = 0; i < ROWS; i++) s += m[i][j];
        printf("%d ", s);
    }
    /* Максимум і його координати */
    int mx = m[0][0], ri = 0, ci = 0;
    for (int i = 0; i < ROWS; i++)
        for (int j = 0; j < COLS; j++)
            if (m[i][j] > mx) { mx = m[i][j]; ri = i; ci = j; }
    printf("\nМакс: %d на позиції [%d][%d]\n", mx, ri, ci);
    /* Транспонування (квадратна матриця) */
    for (int i = 0; i < ROWS; i++)
        for (int j = i+1; j < COLS; j++) {
            int t = m[i][j]; m[i][j] = m[j][i]; m[j][i] = t;
        }
    printf("\nТранспонована:\n");
    print_matrix(m, ROWS, COLS);
    return 0;
}

/* Вправа 4. Решето Ератосфена (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <string.h>
#include <math.h>
#define MAXN 1000
int main(void) {
    console_utf8();
    int n;
    printf("Знайти прості числа до N (≤ %d): ", MAXN);
    scanf("%d", &n);
    char is_prime[MAXN + 1];
    memset(is_prime, 1, sizeof(is_prime));
    is_prime[0] = is_prime[1] = 0;
    /* Решето */
    for (int i = 2; (long long)i*i <= n; i++)
        if (is_prime[i])
            for (int j = i*i; j <= n; j += i)
                is_prime[j] = 0;
    /* Виведення */
    int count = 0;
    long long sum = 0;
    printf("Прості числа до %d:\n", n);
    for (int i = 2; i <= n; i++) {
        if (is_prime[i]) {
            printf("%5d", i);
            count++;
            sum += i;
            if (count % 12 == 0) printf("\n");
        }
    }
    printf("\n");
    printf("Кількість: %d\n", count);
    printf("Сума:      %lld\n", sum);
    printf("Найбільше: ");
    for (int i = n; i >= 2; i--)
        if (is_prime[i]) { printf("%d\n", i); break; }
    return 0;
}

/* Вправа 5. Множення матриць (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#define MAXD 10
void read_matrix(int a[][MAXD], int r, int c, char name) {
    printf("Матриця %c (%dx%d):\n", name, r, c);
    for (int i = 0; i < r; i++)
        for (int j = 0; j < c; j++) {
            printf("  %c[%d][%d] = ", name, i, j);
            scanf("%d", &a[i][j]);
        }
}
void print_matrix(int a[][MAXD], int r, int c) {
    for (int i = 0; i < r; i++) {
        for (int j = 0; j < c; j++) printf("%6d", a[i][j]);
        printf("\n");
    }
}
int main(void) {
    console_utf8();
    int rA, cA, rB, cB;
    printf("Розміри матриці A (рядки стовпці): "); scanf("%d %d",&rA,&cA);
    printf("Розміри матриці B (рядки стовпці): "); scanf("%d %d",&rB,&cB);
    if (cA != rB) {
        printf("Помилка: кількість стовпців A (%d) != рядків B (%d)\n",cA,rB);
        return 1;
    }
    int A[MAXD][MAXD], B[MAXD][MAXD], C[MAXD][MAXD];
    read_matrix(A, rA, cA, 'A');
    read_matrix(B, rB, cB, 'B');
    /* Множення C = A * B */
    for (int i = 0; i < rA; i++)
        for (int j = 0; j < cB; j++) {
            C[i][j] = 0;
            for (int k = 0; k < cA; k++)
                C[i][j] += A[i][k] * B[k][j];
        }
    printf("\nA (%dx%d):\n", rA, cA); print_matrix(A, rA, cA);
    printf("B (%dx%d):\n", rB, cB); print_matrix(B, rB, cB);
    printf("C = A*B (%dx%d):\n", rA, cB); print_matrix(C, rA, cB);
    return 0;
}

2.1 Завдання 19 (таблиця 3.1)

Знайти в масиві пару елементів з найменшою різницею. Вивести пару та різницю.

Основний алгоритм — перебір усіх пар (i < j) з обчисленням |a[i] − a[j]| і запам’ятовуванням мінімуму, складність O(n²). Для контролю копія масиву сортується вставками: у відсортованому масиві найближчі за значенням елементи стоять поруч, тому мінімальна різниця знаходиться за один прохід. Розмір масиву вводиться з клавіатури (2 ≤ n ≤ MAX = 100), заповнення — з клавіатури або випадковими числами.

Рисунок 1 – Блок-схема алгоритму пошуку пари з найменшою різницею

Файл lab07_var19_t1.c
/*
 * Лабораторна робота №7. Індивідуальне завдання №19 (таблиця 3.1).
 * Знайти в масиві пару елементів з найменшою різницею.
 * Вивести пару та різницю.
 *
 * Алгоритм: перебір усіх пар (i < j), різниця |a[i] - a[j]|,
 * запам'ятовується пара з мінімальною різницею. Складність O(n^2).
 * Додатково для перевірки: копію масиву сортуємо вставками - тоді
 * найближча пара обов'язково стоїть поруч, і мінімум шукається за O(n).
 */
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include "../common_console.h"

#define MAX 100

static void print_arr(const char *title, const int a[], int n)
{
    printf("%s[", title);
    for (int i = 0; i < n; i++) printf("%5d", a[i]);
    printf(" ]\n");
}

int main(void)
{
    console_utf8();
    int a[MAX], s[MAX], n, mode;

    printf("Кількість елементів (2..%d): ", MAX);
    if (scanf("%d", &n) != 1 || n < 2 || n > MAX) {
        printf("Помилка: n має бути від 2 до %d\n", MAX);
        return 1;
    }
    printf("Заповнення: 1 - з клавіатури, 2 - випадкові [-50; 50]: ");
    if (scanf("%d", &mode) != 1 || (mode != 1 && mode != 2)) {
        printf("Помилка вибору режиму\n");
        return 1;
    }
    if (mode == 1) {
        printf("Введіть %d цілих чисел: ", n);
        for (int i = 0; i < n; i++)
            if (scanf("%d", &a[i]) != 1) { printf("Помилка введення\n"); return 1; }
    } else {
        srand((unsigned)time(NULL));
        for (int i = 0; i < n; i++) a[i] = rand() % 101 - 50;
    }
    print_arr("Масив до обробки:     ", a, n);

    /* --- основний алгоритм: перебір усіх пар --- */
    int bi = 0, bj = 1;
    int best = abs(a[0] - a[1]);
    for (int i = 0; i < n - 1; i++)
        for (int j = i + 1; j < n; j++) {
            int d = abs(a[i] - a[j]);
            if (d < best) { best = d; bi = i; bj = j; }
        }
    printf("Пара з найменшою різницею: a[%d] = %d, a[%d] = %d\n", bi, a[bi], bj, a[bj]);
    printf("Різниця: %d\n", best);

    /* --- перевірка через відсортовану копію (сортування вставками) --- */
    for (int i = 0; i < n; i++) s[i] = a[i];
    for (int i = 1; i < n; i++) {
        int key = s[i], j = i - 1;
        while (j >= 0 && s[j] > key) { s[j + 1] = s[j]; j--; }
        s[j + 1] = key;
    }
    int best2 = s[1] - s[0];
    for (int i = 1; i < n - 1; i++)
        if (s[i + 1] - s[i] < best2) best2 = s[i + 1] - s[i];
    print_arr("Відсортована копія:   ", s, n);
    printf("Перевірка (сусідні у відсортованому): мінімальна різниця = %d -> %s\n",
           best2, best2 == best ? "збігається" : "ПОМИЛКА");
    print_arr("Масив після обробки:  ", a, n);   /* вихідний масив не змінено */
    return 0;
}

2.2 Завдання 19 (таблиця 3.2)

Переставити елементи масиву так, щоб парні стояли перед непарними, зберігаючи відносний порядок.

Застосовано стабільний алгоритм на місці: кожен знайдений парний елемент вставляється одразу після вже зібраних парних, а елементи між ними зсуваються вправо на одну позицію (як у сортуванні вставками). Відносний порядок зберігається і серед парних, і серед непарних. Складність O(n²), додаткова пам’ять O(1). Перевірка парності a[i] % 2 == 0 коректна і для від’ємних чисел.

Рисунок 2 – Блок-схема стабільного переставлення парних елементів

Файл lab07_var19_t2.c
/*
 * Лабораторна робота №7. Індивідуальне завдання №19 (таблиця 3.2).
 * Дано масив. Переставити елементи так, щоб парні стояли перед
 * непарними, зберігаючи відносний порядок.
 *
 * Алгоритм (стабільний, на місці): прохід зліва направо; кожен
 * знайдений парний елемент "вставляється" одразу після вже
 * зібраних парних - елементи між ними зсуваються вправо на 1
 * (як у сортуванні вставками). Відносний порядок і парних, і
 * непарних зберігається. Складність O(n^2), додаткова пам'ять O(1).
 */
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include "../common_console.h"

#define MAX 100

static void print_arr(const char *title, const int a[], int n)
{
    printf("%s[", title);
    for (int i = 0; i < n; i++) printf("%5d", a[i]);
    printf(" ]\n");
}

int main(void)
{
    console_utf8();
    int a[MAX], n, mode;

    printf("Кількість елементів (1..%d): ", MAX);
    if (scanf("%d", &n) != 1 || n < 1 || n > MAX) {
        printf("Помилка: n має бути від 1 до %d\n", MAX);
        return 1;
    }
    printf("Заповнення: 1 - з клавіатури, 2 - випадкові [-50; 50]: ");
    if (scanf("%d", &mode) != 1 || (mode != 1 && mode != 2)) {
        printf("Помилка вибору режиму\n");
        return 1;
    }
    if (mode == 1) {
        printf("Введіть %d цілих чисел: ", n);
        for (int i = 0; i < n; i++)
            if (scanf("%d", &a[i]) != 1) { printf("Помилка введення\n"); return 1; }
    } else {
        srand((unsigned)time(NULL));
        for (int i = 0; i < n; i++) a[i] = rand() % 101 - 50;
    }
    print_arr("До обробки:    ", a, n);

    int k = 0;                      /* кількість уже переставлених парних */
    for (int i = 0; i < n; i++) {
        if (a[i] % 2 == 0) {        /* працює і для від'ємних: -4 % 2 == 0 */
            int t = a[i];
            for (int j = i; j > k; j--) a[j] = a[j - 1];   /* зсув вправо */
            a[k] = t;
            k++;
        }
    }
    print_arr("Після обробки: ", a, n);
    printf("Парних: %d, непарних: %d\n", k, n - k);
    return 0;
}

Висновки

Засвоєно оголошення, ініціалізацію і обробку одновимірних та двовимірних масивів, алгоритми сортування (бульбашкою, вставками), лінійного і бінарного пошуку, решета Ератосфена, множення матриць. Для варіанта 19 реалізовано пошук пари з найменшою різницею (з перевіркою через сортування) і стабільне переставлення парних елементів перед непарними; програми протестовано на ручних і випадкових даних та на некоректному розмірі.

Відповіді на контрольні питання

1. Що таке масив у мові С? Як оголосити масив з N елементів типу double? Послідовність однотипних елементів у суміжних комірках пам’яті: double a[N]; (N — константа).

2. Яким є діапазон допустимих індексів масиву розміром N? Що відбувається при виході за межі? Від 0 до N−1. Вихід за межі не перевіряється — це невизначена поведінка: пошкодження сусідніх даних або аварійне завершення.

3. Як ініціалізувати масив нулями одним рядком коду? int a[N] = {0};

4. Опишіть алгоритм сортування методом бульбашки. Яка його часова складність? Порівнюються сусідні елементи і міняються місцями, якщо стоять неправильно; за кожен прохід найбільший елемент «спливає» в кінець. O(n²), у кращому випадку з прапорцем обмінів — O(n).

5. Яка умова обов’язкова для бінарного пошуку? Скільки кроків максимально для масиву з 1000 елементів? Масив має бути відсортований. Не більше ⌈log₂(1001)⌉ = 10 порівнянь.

6. Як оголосити двовимірний масив 5×7? Як отримати доступ до елемента в рядку 2, стовпці 4? int m[5][7];, елемент — m[2][4] (нумерація з 0).

7. Що означає row-major order? Як двовимірний масив зберігається в пам’яті? Зберігання по рядках: спочатку всі елементи рядка 0, потім рядка 1 тощо; адреса m[i][j] = база + (i·COLS + j)·sizeof(елемента).

8. Як транспонувати квадратну матрицю на місці? Чому перебираємо лише j > i? Міняти місцями a[i][j] і a[j][i] для j > i; якщо перебирати всі j, кожну пару буде обміняно двічі і матриця повернеться до початкового стану.

9. Поясніть алгоритм решета Ератосфена. Чому закреслення починають з i²? Для кожного ще не закресленого i закреслюються його кратні; менші кратні i·k (k < i) уже закреслені при обробці меншого дільника k.

10. Що таке складність алгоритму множення матриць? Напишіть три вкладені цикли для множення A(M×K) * B(K×N). O(M·K·N): for(i=0;i<M;i++) for(j=0;j<N;j++){C[i][j]=0; for(k=0;k<K;k++) C[i][j]+=A[i][k]*B[k][j];}
