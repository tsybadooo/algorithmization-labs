int  is_prime(int n);

int  lcm(int a, int b);
long long factorial(int n);
int  sum_digits(int n);

int is_prime(int n) {
    if (n < 2) return 0;
    if (n == 2) return 1;
    if (n % 2 == 0) return 0;
    for (int i = 3; (long long)i*i <= n; i += 2)
        if (n % i == 0) return 0;
    return 1;
}
int gcd(int a, int b) {
    return b == 0 ? a : gcd(b, a % b);
}
int lcm(int a, int b) {
    return a / gcd(a, b) * b;  /* ділення ПЕРЕД множенням — уникаємо переповнення */
}
long long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}
int sum_digits(int n) {
    int s = 0;
    if (n < 0) n = -n;
    if (n == 0) return 0;
    while (n > 0) { s += n % 10; n /= 10; }
    return s;
}
int count_digits(int n) {
    if (n == 0) return 1;
    int c = 0;
    if (n < 0) n = -n;
    while (n > 0) { c++; n /= 10; }
    return c;
}
int main(void) {
    console_utf8();
    /* Прості числа */
    printf("Прості від 2 до 50: ");
    for (int i = 2; i <= 50; i++)
        if (is_prime(i)) printf("%d ", i);
    printf("\n");
    /* НСД і НСК */
    int a = 48, b = 36;
    printf("НСД(%d,%d)=%d  НСК=%d\n", a, b, gcd(a,b), lcm(a,b));
    /* Факторіали */
    printf("\nТаблиця факторіалів:\n");
    for (int n = 0; n <= 12; n++)
        printf("  %2d! = %lld\n", n, factorial(n));
    /* Цифри числа */
    int num = 123456;
    printf("\nЧисло: %d\n", num);
    printf("Кількість цифр: %d\n", count_digits(num));
    printf("Сума цифр:      %d\n", sum_digits(num));
    return 0;
}

/* Вправа 2. Рекурсія проти ітерації — числа Фібоначчі (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
long long call_count;  /* лічильник викликів */
/* Рекурсивний — O(2^n) */
long long fib_rec(int n) {
    call_count++;
    if (n <= 1) return n;
    return fib_rec(n-1) + fib_rec(n-2);
}
/* Ітеративний — O(n) */
long long fib_iter(int n) {
    if (n <= 1) return n;
    long long a = 0, b = 1;
    for (int i = 2; i <= n; i++) {
        long long c = a + b; a = b; b = c;
    }
    return b;
}
/* Мемоізований — O(n) через кеш */
long long memo[50];
long long fib_memo(int n) {
    call_count++;
    if (n <= 1) return n;
    if (memo[n] != -1) return memo[n];
    memo[n] = fib_memo(n-1) + fib_memo(n-2);
    return memo[n];
}
int main(void) {
    console_utf8();
    printf("%-4s %-12s %-12s %-14s %-12s\n",
           "n", "fib(n)", "рекурс.вик.", "мемо.вик.", "ітер.");
    printf("%s\n", "-------------------------------------------------------");
    for (int n = 0; n <= 20; n++) {
        call_count = 0;
        long long r = fib_rec(n);
        long long rc = call_count;
        for (int i = 0; i < 50; i++) memo[i] = -1;
        call_count = 0;
        long long m = fib_memo(n);
        long long mc = call_count;
        long long it = fib_iter(n);
        if (m != r || it != r) printf("Розбіжність результатів!\n");
        printf("%-4d %-12lld %-14lld %-14lld %-12lld\n",
               n, r, rc, mc, it);
    }
    return 0;
}

/* Вправа 3. Ханойські вежі та рекурсивний відлік (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
int total_moves;
void hanoi(int disks, char from, char to, char via) {
    if (disks == 1) {
        printf("  Диск 1: %c -> %c\n", from, to);
        total_moves++;
        return;
    }
    hanoi(disks-1, from, via, to);   /* перемістити n-1 на допоміжний */
    printf("  Диск %d: %c -> %c\n", disks, from, to);
    total_moves++;
    hanoi(disks-1, via, to, from);   /* перемістити n-1 з допоміжного */
}
int main(void) {
    console_utf8();
    printf("=== Ханойські вежі ===\n");
    for (int n = 1; n <= 4; n++) {
        total_moves = 0;
        printf("\n%d дисків: (A->C через B)\n", n);
        hanoi(n, 'A', 'C', 'B');
        printf("  Ходів: %d  (2^%d-1=%d)\n",
               total_moves, n, (1<<n)-1);
    }
    /* Рекурсивний зворотній відлік */
    printf("\n=== Рекурсивний відлік від 5 до 0 ===\n");
    /* Функція countdown через лямбду-ідею у С — вкладений виклик */
    void countdown(int n) {
        if (n < 0) return;
        printf("%d ", n);
        countdown(n - 1);
    }
    countdown(5);
    printf("\n");
    return 0;
}

/* Вправа 4. Функції для масивів — обробка через вказівники (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <math.h>
/* Бібліотека функцій для масивів */
void   arr_print(const int *a, int n, const char *label);
int    arr_sum(const int *a, int n);
double arr_avg(const int *a, int n);
int    arr_min(const int *a, int n);
int    arr_max(const int *a, int n);
int    arr_count_if(const int *a, int n, int (*pred)(int));
void   arr_sort_bubble(int *a, int n);
void arr_print(const int *a, int n, const char *label) {
    printf("%s [ ", label);
    for (int i = 0; i < n; i++) printf("%d ", a[i]);
    printf("]\n");
}
int arr_sum(const int *a, int n) {
    int s = 0;
    for (int i = 0; i < n; i++) s += a[i];
    return s;
}
double arr_avg(const int *a, int n) {
    return n > 0 ? (double)arr_sum(a, n) / n : 0.0;
}
int arr_min(const int *a, int n) {
    int m = a[0];
    for (int i = 1; i < n; i++) if (a[i] < m) m = a[i];
    return m;
}
int arr_max(const int *a, int n) {
    int m = a[0];
    for (int i = 1; i < n; i++) if (a[i] > m) m = a[i];
    return m;
}
/* Підрахувати елементи, що задовольняють предикату */
int arr_count_if(const int *a, int n, int (*pred)(int)) {
    int c = 0;
    for (int i = 0; i < n; i++) if (pred(a[i])) c++;
    return c;
}
void arr_sort_bubble(int *a, int n) {
    for (int i = 0; i < n-1; i++)
        for (int j = 0; j < n-1-i; j++)
            if (a[j] > a[j+1]) {
                int t = a[j]; a[j] = a[j+1]; a[j+1] = t;
            }
}
/* Предикати */
int is_even(int x)     { return x % 2 == 0; }
int is_positive(int x) { return x > 0; }
int is_prime_p(int x)  {
    if (x < 2) return 0;
    for (int i = 2; (long long)i*i <= x; i++)
        if (x % i == 0) return 0;
    return 1;
}
int main(void) {
    console_utf8();
    int arr[] = {7, -3, 15, 2, 11, -8, 4, 23, 6, 1};
    int n = 10;
    arr_print(arr, n, "Масив:");
    printf("Сума: %d\n", arr_sum(arr, n));
    printf("Середнє: %.2f\n", arr_avg(arr, n));
    printf("Мін: %d, Макс: %d\n", arr_min(arr, n), arr_max(arr, n));
    printf("Парних: %d\n",   arr_count_if(arr, n, is_even));
    printf("Додатніх: %d\n", arr_count_if(arr, n, is_positive));
    printf("Простих: %d\n",  arr_count_if(arr, n, is_prime_p));
    arr_sort_bubble(arr, n);
    arr_print(arr, n, "Відсортований:");
    return 0;
}

2 Індивідуальні завдання (варіант 19)

У методичних вказівках індивідуальні завдання подано у двох таблицях («Варіант 1» і «Варіант 2»), кожна з яких містить 20 завдань. Оскільки з вказівок не однозначно випливає, яку таблицю обирати, виконано завдання №19 з обох таблиць.

2.1 Завдання 19 (таблиця 3.1)

Написати функцію rotate_digits(n, k) — циклічно зсунути цифри числа на k позицій. Написати sum_of_proper_divisors(n).

Для числа з L цифр зсув уліво на k (k нормалізується до 0 ≤ k < L, від’ємне k означає зсув управо) обчислюється без перетворення в рядок: перші k цифр head = n / 10L−k, решта tail = n mod 10L−k, результат tail·10k + head. Рекурсивна версія зводить зсув на k до зсуву на 1 і зсуву на k − 1 (базовий випадок k = 0). Сума власних дільників шукається перебором до √n: дільники утворюють пари i та n/i. Додатково визначається тип числа: досконале, надлишкове чи недостатнє.

Рисунок 1 – Блок-схема функції rotate_digits

Файл lab08_var19_t1.c
/*
 * Лабораторна робота №8. Індивідуальне завдання №19 (таблиця 3.1).
 * 1) rotate_digits(n, k) - циклічно зсунути цифри числа на k позицій
 *    (k > 0 - вліво, k < 0 - вправо). Приклад: 12345, k=2 -> 34512.
 * 2) sum_of_proper_divisors(n) - сума власних дільників числа n
 *    (усіх дільників, менших за n). Приклад: 28 -> 1+2+4+7+14 = 28.
 * Для зсуву написано ітеративну і рекурсивну версії та порівняно їх.
 */
#include <stdio.h>
#include "../common_console.h"

/* ---------- прототипи ---------- */
int  count_digits(long n);
long ipow10(int p);
long rotate_digits(long n, int k);
long rotate_left_once(long n, int len);
long rotate_digits_rec(long n, int k, int len);
long sum_of_proper_divisors(long n);
void print_divisors(long n);

int main(void)
{
    console_utf8();
    long n;
    int k;
    printf("Введіть натуральне число n та зсув k: ");
    if (scanf("%ld %d", &n, &k) != 2 || n <= 0) {
        printf("Помилка: n має бути натуральним, k - цілим\n");
        return 1;
    }
    int len = count_digits(n);
    int kk = ((k % len) + len) % len;           /* нормалізований зсув вліво */

    long r1 = rotate_digits(n, k);
    long r2 = rotate_digits_rec(n, kk, len);
    printf("Кількість цифр: %d\n", len);
    printf("rotate_digits(%ld, %d)           = %ld\n", n, k, r1);
    printf("рекурсивна версія (вліво на %d)   = %ld  -> %s\n", kk, r2,
           r1 == r2 ? "збігається" : "РОЗБІЖНІСТЬ");

    long s = sum_of_proper_divisors(n);
    printf("Власні дільники %ld: ", n);
    print_divisors(n);
    printf("sum_of_proper_divisors(%ld) = %ld", n, s);
    if (s == n)      printf("  (досконале число)\n");
    else if (s > n)  printf("  (надлишкове число)\n");
    else             printf("  (недостатнє число)\n");
    return 0;
}

/* Призначення: кількість десяткових цифр числа.
   Параметри: n - невід'ємне ціле. Повертає: кількість цифр (для 0 - 1). */
int count_digits(long n)
{
    int c = 1;
    while (n >= 10) { n /= 10; c++; }
    return c;
}

/* Повертає 10 у степені p (p >= 0) без використання pow(). */
long ipow10(int p)
{
    long r = 1;
    while (p-- > 0) r *= 10;
    return r;
}

/* Призначення: циклічний зсув цифр (ітеративно, за формулою).
   Параметри: n - натуральне число, k - зсув (k>0 вліво, k<0 вправо).
   Повертає: число після зсуву. Ведучі нулі, що з'являються
   (напр. 1002 -> 0021), за правилами запису числа відкидаються. */
long rotate_digits(long n, int k)
{
    int len = count_digits(n);
    k = ((k % len) + len) % len;          /* зсув вправо = зсув вліво на len-k */
    long p = ipow10(len - k);
    long head = n / p;                    /* перші k цифр      */
    long tail = n % p;                    /* останні len-k цифр */
    return tail * ipow10(k) + head;
}

/* Зсув на 1 позицію вліво з урахуванням початкової довжини len. */
long rotate_left_once(long n, int len)
{
    long p = ipow10(len - 1);
    return (n % p) * 10 + n / p;
}

/* Рекурсивна версія: зсув на k = зсув на 1 + зсув на (k-1).
   База рекурсії: k == 0. Глибина рекурсії - k (k < len). */
long rotate_digits_rec(long n, int k, int len)
{
    if (k == 0) return n;
    return rotate_digits_rec(rotate_left_once(n, len), k - 1, len);
}

/* Призначення: сума власних дільників числа (дільники < n).
   Перебір до sqrt(n): дільники йдуть парами i та n/i. */
long sum_of_proper_divisors(long n)
{
    if (n == 1) return 0;
    long s = 1;
    for (long i = 2; i * i <= n; i++)
        if (n % i == 0) {
            s += i;
            if (i != n / i) s += n / i;
        }
    return s;
}

/* Виводить усі власні дільники числа у порядку зростання. */
void print_divisors(long n)
{
    int any = 0;
    for (long i = 1; i < n; i++)
        if (n % i == 0) { printf("%s%ld", any ? " " : "", i); any = 1; }
    if (!any) printf("(немає)");
    printf("\n");
}

/* Повертає 1, якщо рік високосний: ділиться на 4, але не на 100,
   або ділиться на 400. Інакше 0. */
int is_leap_year(int year)
{
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

/* Повертає кількість днів у місяці month (1..12) року year,
   або 0, якщо номер місяця некоректний. */
int days_in_month(int month, int year)
{
    static const int dm[12] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
    if (month < 1 || month > 12) return 0;
    if (month == 2 && is_leap_year(year)) return 29;
    return dm[month - 1];
}

/* Перевірка існування дати (рік >= 1). Повертає 1 - коректна, 0 - ні. */
int is_valid_date(int day, int month, int year)
{
    return year >= 1 && month >= 1 && month <= 12 &&
           day >= 1 && day <= days_in_month(month, year);
}

/* Ітеративно: сума днів повних попередніх місяців + день. */
int day_of_year(int day, int month, int year)
{
    int n = day;
    for (int m = 1; m < month; m++) n += days_in_month(m, year);
    return n;
}

/* Рекурсивно: day_of_year(d, m) = days_in_month(m-1) + day_of_year(d, m-1);
   база: m == 1 -> d. */
int day_of_year_rec(int day, int month, int year)
{
    if (month == 1) return day;
    return days_in_month(month - 1, year) + day_of_year_rec(day, month - 1, year);
}

Висновки

Засвоєно оголошення прототипів, визначення і виклик функцій, передачу параметрів за значенням і через вказівники, рекурсію та її порівняння з ітерацією. Усі індивідуальні функції не використовують глобальних змінних, мають прототипи і протестовані щонайменше на трьох наборах даних, у т.ч. некоректних.

Відповіді на контрольні питання

1. Що таке функція в С? Які обов’язкові складові визначення функції? Іменований блок коду, що виконує задачу і може викликатися багаторазово. Складові: тип результату, ім’я, список параметрів, тіло.

2. Що таке прототип функції і навіщо він потрібен? Що буде, якщо не вказати прототип? Оголошення функції до її використання (тип, ім’я, типи параметрів) — дає компілятору змогу перевіряти виклики. Без прототипу в C11 виклик невідомої функції є помилкою/попередженням, а в старих стандартах вважалось, що вона повертає int, що могло давати хибні результати.

3. Поясніть різницю між передачею параметра за значенням та через вказівник. За значенням функція отримує копію і не змінює оригінал: void f(int x); через вказівник — адресу і може змінити змінну: void g(int *x){ *x = 5; }, виклик g(&a).

4. Що таке локальні та глобальні змінні? Чому зловживати глобальними змінними небажано? Локальні видимі в блоці, глобальні — у всьому файлі/програмі. Глобальні створюють приховані залежності, ускладнюють тестування і повторне використання.

5. Що таке статична локальна змінна (static)? Локальна за видимістю, але існує весь час роботи програми і зберігає значення між викликами; ініціалізується один раз.

6. Що таке рекурсія? Що таке базовий випадок і чому він обов’язковий? Виклик функцією самої себе. Базовий випадок припиняє рекурсію; без нього виклики тривають до переповнення стека.

7. Порівняйте рекурсивне та ітеративне обчислення чисел Фібоначчі за складністю O(). Наївна рекурсія — O(φⁿ) ≈ O(1,618ⁿ) викликів, ітерація — O(n) і O(1) пам’яті, мемоїзація — O(n).

8. Що таке функція типу void? Чи може void-функція мати оператор return? Функція, що не повертає значення; може містити return; без виразу для дострокового виходу.

9. Як повернути кілька значень із функції? Через параметри-вказівники, структуру як результат або масив, переданий у функцію.
