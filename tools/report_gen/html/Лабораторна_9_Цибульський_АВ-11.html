/* Сума через арифметику вказівників */
int sum_ptr(const int *arr, int n) {
    int s = 0;
    const int *end = arr + n;
    while (arr < end) s += *arr++;
    return s;
}
/* Повернути вказівник на максимум */
int *find_max(int *arr, int n) {
    int *mx = arr;
    for (int i = 1; i < n; i++)
        if (*(arr + i) > *mx) mx = arr + i;
    return mx;
}

/* Вправа 2. Динамічний масив з автозбільшенням (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <stdlib.h>
int main(void) {
    console_utf8();
    int *arr     = NULL;
    int  size     = 0;
    int  capacity = 0;
    printf("Вводьте числа (0 — кінець):\n");
    int x;
    while (scanf("%d", &x) == 1 && x != 0) {
        /* Збільшити capacity якщо потрібно */
        if (size >= capacity) {
            int new_cap = capacity == 0 ? 4 : capacity * 2;
            int *tmp = (int *)realloc(arr, new_cap * sizeof(int));
            if (tmp == NULL) {
                printf("Помилка пам'яті!\n");
                free(arr);
                return 1;
            }
            arr = tmp;
            capacity = new_cap;
            printf("  [realloc: capacity=%d]\n", capacity);
        }
        arr[size++] = x;
    }
    printf("\nЗбережено %d чисел (capacity=%d):\n", size, capacity);
    for (int i = 0; i < size; i++) printf("%d ", arr[i]);
    printf("\n");
    /* Знайти мін та макс через вказівники */
    if (size > 0) {
        int *mn = arr, *mx = arr;
        for (int *p = arr + 1; p < arr + size; p++) {
            if (*p < *mn) mn = p;
            if (*p > *mx) mx = p;
        }
        printf("Мін=%d [%td]  Макс=%d [%td]\n",
               *mn, mn-arr, *mx, mx-arr);
    }
    free(arr);
    arr = NULL;
    return 0;
}

/* Вправа 3. const-вказівники та передача масивів (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <string.h>
/* const int* — не можна змінити *p (захист даних) */
int sum_const(const int *arr, int n) {
    int s = 0;
    for (int i = 0; i < n; i++) {
        s += arr[i];
        /* arr[i] = 0; -- ПОМИЛКА компіляції */
    }
    return s;
}
/* Копіювання рядка без strcpy */
char *my_strcpy(char *dst, const char *src) {
    char *d = dst;
    while ((*d++ = *src++) != '\0');
    return dst;
}
/* Довжина рядка без strlen */
size_t my_strlen(const char *s) {
    const char *p = s;
    while (*p) p++;
    return p - s;
}
int main(void) {
    console_utf8();
    int arr[] = {5, 3, 8, 1, 9};
    int n = 5;
    printf("Сума = %d\n", sum_const(arr, n));
    /* const int * vs int * const */
    int x = 10, y = 20;
    const int *p1 = &x;  /* не можна: *p1 = 5; */
    int * const p2 = &x; /* не можна: p2 = &y; */
    printf("p1 вказує на %d\n", *p1);
    p1 = &y;    /* можна змінити адресу */
    printf("p1 тепер вказує на %d\n", *p1);
    *p2 = 99;   /* можна змінити значення */
    printf("x = %d\n", x);
    /* Рядкові функції */
    char src[] = "Hello World";
    char dst[50];
    my_strcpy(dst, src);
    printf("Копія: %s\n", dst);
    printf("Довжина: %zu  (strlen: %zu)\n",
           my_strlen(src), strlen(src));
    return 0;
}

/* Вправа 4. Сортування через вказівник на функцію (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <stdlib.h>
/* Компаратори */
int cmp_asc(const void *a, const void *b) {
    return *(int*)a - *(int*)b;
}
int cmp_desc(const void *a, const void *b) {
    return *(int*)b - *(int*)a;
}
int cmp_abs(const void *a, const void *b) {
    int va = *(int*)a < 0 ? -*(int*)a : *(int*)a;
    int vb = *(int*)b < 0 ? -*(int*)b : *(int*)b;
    return va - vb;
}
void print_arr(const int *arr, int n) {
    for (int i = 0; i < n; i++) printf("%4d", arr[i]);
    printf("\n");
}
int main(void) {
    console_utf8();
    int arr[] = {5, -3, 8, -1, 9, -7, 2, 4};
    int n = 8;
    int tmp[8];
    printf("Оригінал:    "); print_arr(arr, n);
    for (int i = 0; i < n; i++) tmp[i] = arr[i];
    qsort(tmp, n, sizeof(int), cmp_asc);
    printf("Зростання:   "); print_arr(tmp, n);
    for (int i = 0; i < n; i++) tmp[i] = arr[i];
    qsort(tmp, n, sizeof(int), cmp_desc);
    printf("Спадання:    "); print_arr(tmp, n);
    for (int i = 0; i < n; i++) tmp[i] = arr[i];
    qsort(tmp, n, sizeof(int), cmp_abs);
    printf("За модулем:  "); print_arr(tmp, n);
    return 0;
}

/* Вправа 5. Динамічна матриця та операції (з методичних вказівок) */
#include "../common_console.h"
#include <stdio.h>
#include <stdlib.h>
int **alloc_matrix(int n) {
    int **m = (int **)malloc(n * sizeof(int *));
    if (!m) return NULL;
    for (int i = 0; i < n; i++) {
        m[i] = (int *)malloc(n * sizeof(int));
        if (!m[i]) {
            for (int j = 0; j < i; j++) free(m[j]);
            free(m); return NULL;
        }
    }
    return m;
}
void free_matrix(int **m, int n) {
    for (int i = 0; i < n; i++) free(m[i]);
    free(m);
}
void print_matrix(int **m, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) printf("%5d", m[i][j]);
        printf("\n");
    }
}
int main(void) {
    console_utf8();
    int n = 4;
    int **mat = alloc_matrix(n);
    if (!mat) { perror("alloc"); return 1; }
    /* Заповнити */
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            mat[i][j] = i * n + j + 1;
    printf("Матриця:\n"); print_matrix(mat, n);
    /* Суми діагоналей */
    int d1 = 0, d2 = 0;
    for (int i = 0; i < n; i++) {
        d1 += mat[i][i];
        d2 += mat[i][n-1-i];
    }
    printf("Головна діагональ: %d\n", d1);
    printf("Побічна діагональ: %d\n", d2);
    /* Транспонування */
    for (int i = 0; i < n; i++)
        for (int j = i + 1; j < n; j++) {
            int t = mat[i][j]; mat[i][j] = mat[j][i]; mat[j][i] = t;
        }
    printf("\nТранспонована:\n"); print_matrix(mat, n);
    free_matrix(mat, n);
    return 0;
}

2 Індивідуальне завдання (варіант 19)

Написати функцію matrix_mul(int **a, int **b, int **c, int n) — множення квадратних матриць через вказівники.

Матриці подано масивом вказівників на рядки (int **). Функція alloc_matrix виділяє масив вказівників (malloc) і рядки (calloc), перевіряє кожен результат на NULL і в разі помилки звільняє вже виділені рядки. Функція free_matrix приймає int *** — це дає змогу після free() встановити вказівник викликача в NULL. У matrix_mul усі звертання виконуються через арифметику вказівників *(*(a + i) + k); складність O(n³).

Рисунок 1 – Блок-схема функції matrix_mul

Файл lab09_var19.c
/*
 * Лабораторна робота №9. Індивідуальне завдання №19.
 * Написати функцію matrix_mul(int **a, int **b, int **c, int n) -
 * множення квадратних матриць через вказівники.
 *
 * Матриці зберігаються як масив вказівників на рядки (int **),
 * пам'ять виділяється динамічно; результат malloc перевіряється на NULL,
 * усі блоки звільняються, після free() вказівник обнуляється.
 * Доступ до елементів - через арифметику вказівників: *(*(a+i)+k).
 */
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#include "../common_console.h"

/* ---------- прототипи ---------- */
int **alloc_matrix(int n);
void free_matrix(int ***pm, int n);
void fill_matrix(int **m, int n, int mode);
void print_matrix(const char *name, int **m, int n);
void matrix_mul(int **a, int **b, int **c, int n);

int main(void)
{
    console_utf8();
    int n, mode;
    printf("Розмір квадратних матриць n (1..20): ");
    if (scanf("%d", &n) != 1 || n < 1 || n > 20) {
        printf("Помилка: n має бути від 1 до 20\n");
        return 1;
    }
    printf("Заповнення: 1 - з клавіатури, 2 - випадкові [-9; 9]: ");
    if (scanf("%d", &mode) != 1 || (mode != 1 && mode != 2)) {
        printf("Помилка вибору режиму\n");
        return 1;
    }
    srand((unsigned)time(NULL));

    int **a = alloc_matrix(n), **b = alloc_matrix(n), **c = alloc_matrix(n);
    if (a == NULL || b == NULL || c == NULL) {
        printf("Помилка: недостатньо пам'яті\n");
        free_matrix(&a, n); free_matrix(&b, n); free_matrix(&c, n);
        return 1;
    }
    if (mode == 1) printf("Введіть матрицю A (%d x %d):\n", n, n);
    fill_matrix(a, n, mode);
    if (mode == 1) printf("Введіть матрицю B (%d x %d):\n", n, n);
    fill_matrix(b, n, mode);

    matrix_mul(a, b, c, n);
    print_matrix("A", a, n);
    print_matrix("B", b, n);
    print_matrix("C = A * B", c, n);

    free_matrix(&a, n); free_matrix(&b, n); free_matrix(&c, n);
    printf("Пам'ять звільнено: a=%p b=%p c=%p\n", (void *)a, (void *)b, (void *)c);
    return 0;
}

/* Виділяє матрицю n x n як масив вказівників на рядки.
   Повертає вказівник або NULL (у разі помилки вже виділене звільняється). */
int **alloc_matrix(int n)
{
    int **m = (int **)malloc((size_t)n * sizeof(int *));
    if (m == NULL) return NULL;
    for (int i = 0; i < n; i++) {
        m[i] = (int *)calloc((size_t)n, sizeof(int));
        if (m[i] == NULL) {                    /* відкат: звільняємо попередні рядки */
            for (int j = 0; j < i; j++) free(m[j]);
            free(m);
            return NULL;
        }
    }
    return m;
}

/* Звільняє матрицю і встановлює вказівник у NULL (тому передається int ***). */
void free_matrix(int ***pm, int n)
{
    if (pm == NULL || *pm == NULL) return;
    for (int i = 0; i < n; i++) { free((*pm)[i]); (*pm)[i] = NULL; }
    free(*pm);
    *pm = NULL;
}

/* Заповнює матрицю: mode 1 - з клавіатури, mode 2 - випадковими числами. */
void fill_matrix(int **m, int n, int mode)
{
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++) {
            if (mode == 1) {
                if (scanf("%d", *(m + i) + j) != 1) *(*(m + i) + j) = 0;
            } else {
                *(*(m + i) + j) = rand() % 19 - 9;
            }
        }
}

/* Друк матриці з вирівнюванням стовпців. */
void print_matrix(const char *name, int **m, int n)
{
    printf("%s:\n", name);
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) printf("%6d", *(*(m + i) + j));
        printf("\n");
    }
}

/* Множення квадратних матриць: c[i][j] = сума a[i][k]*b[k][j], k=0..n-1.
   Усі звертання - через вказівники. Складність O(n^3). */
void matrix_mul(int **a, int **b, int **c, int n)
{
    for (int i = 0; i < n; i++) {
        int *ci = *(c + i);               /* вказівник на i-й рядок результату */
        for (int j = 0; j < n; j++) {
            int s = 0;
            for (int k = 0; k < n; k++)
                s += *(*(a + i) + k) * *(*(b + k) + j);
            *(ci + j) = s;
        }
    }
}

Висновки

Засвоєно операції з вказівниками (&, *, арифметика), динамічне виділення пам’яті функціями malloc, calloc, realloc і звільнення free, обробку помилок виділення, роботу з динамічними двовимірними масивами. Індивідуальна функція множення матриць через вказівники протестована на трьох розмірах даних.

Відповіді на контрольні питання

1. Що таке вказівник? Чим відрізняється int x та int *p? Змінна, що зберігає адресу іншої змінної. x зберігає ціле значення, p — адресу цілого.

2. Що роблять оператори & та * стосовно вказівників? & — отримання адреси змінної, * — розіменування (доступ до значення за адресою).

3. Що таке NULL-вказівник? Чому перед розіменуванням потрібна перевірка на NULL? Вказівник, що нікуди не вказує. Розіменування NULL — невизначена поведінка (зазвичай аварійне завершення), тому результат malloc та функцій пошуку треба перевіряти.

4. Що таке арифметика вказівників? Що означає ptr+1 для int* і для char*? Додавання/віднімання цілих зсуває вказівник на кількість елементів: для int* — на sizeof(int) (4 байти), для char* — на 1 байт.

5. Чим відрізняються malloc(), calloc() та realloc()? malloc виділяє неініціалізований блок; calloc — блок для n елементів, заповнений нулями; realloc змінює розмір виділеного блоку (може перемістити його).

6. Що таке витік пам’яті (memory leak)? Як його уникнути? Виділена пам’ять, на яку втрачено вказівник і яку не звільнено. Кожному malloc — свій free, звільнення на всіх шляхах виходу, перевірка інструментами (Valgrind).

7. Що таке висячий вказівник (dangling pointer)? Як він виникає? Вказівник на вже звільнену або неіснуючу пам’ять — після free() чи повернення адреси локальної змінної. Запобігання: присвоювати NULL після free.

8. Поясніть різницю між const int *p і int * const p. const int *p — не можна змінювати значення через p, але можна змінити сам p; int * const p — сам вказівник незмінний, значення змінювати можна.

9. Як виділити та правильно звільнити пам’ять для двовимірного масиву N*M? Виділити масив N вказівників, потім кожен рядок з M елементів; звільняти в зворотному порядку — спочатку рядки, потім масив вказівників.

10. Чому sizeof(arr)/sizeof(arr[0]) не працює для параметра-масиву? Параметр-масив перетворюється на вказівник, тож sizeof(arr) дає розмір вказівника, а не масиву; розмір потрібно передавати окремим параметром.
